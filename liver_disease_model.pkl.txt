function predictLiver() {
    const formData = {
        Total_Bilirubin: parseFloat(document.getElementById("Total_Bilirubin").value),
        Direct_Bilirubin: parseFloat(document.getElementById("Direct_Bilirubin").value),
        Alkaline_Phosphotase: parseInt(document.getElementById("Alkaline_Phosphotase").value),
        Alamine_Aminotransferase: parseInt(document.getElementById("Alamine_Aminotransferase").value),
        Total_Proteins: parseFloat(document.getElementById("Total_Proteins").value),
        Albumin: parseFloat(document.getElementById("Albumin").value),
        Albumin_and_Globulin_Ratio: parseFloat(document.getElementById("Albumin_and_Globulin_Ratio").value)
    };

    fetch("http://127.0.0.1:5000/predict", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(formData)
    })
    .then(res => res.json())
    .then(data => {
        if (data.error) throw new Error(data.error);
        const resultDiv = document.getElementById("predictionResult");
        const riskLevel = data.prediction === 1 ? "HIGH RISK" : "NORMAL";
        const riskClass = data.prediction === 1 ? "risk-high" : "risk-low";
        resultDiv.innerHTML = `
            <p class="lead ${riskClass}">Prediction Result: ${riskLevel}</p>
            <p>Probability of Liver Disease: ${data.probability}%</p>
        `;
        document.getElementById("assessmentResult").style.display = "block";
        document.getElementById("assessmentResult").scrollIntoView({ behavior: 'smooth' });
    })
    .catch(err => alert("Error: " + err.message));
}
